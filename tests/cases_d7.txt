# warp trace: identity/affine reprojects (exact kernel scales)

--config CPL_DEBUG=ON raster reproject --dst-crs EPSG:4326 --dst-nodata 7 g1.tif d1.tif
--config CPL_DEBUG=ON raster reproject --dst-crs EPSG:4326 --dst-nodata 7,9 mb3.tif d1.tif

--config CPL_DEBUG=ON raster reproject --dst-crs EPSG:4326 --size 256,128 big.tif d1.tif
