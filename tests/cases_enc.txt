# write: ENCODING lco / SHAPE_ENCODING config
vector convert fv.json w.shp
vector convert --lco ENCODING=UTF-8 fv.json w.shp
vector convert --lco ENCODING=ISO-8859-1 fv.json w.shp
vector convert --lco ENCODING=CP1251 fv.json w.shp
vector convert --lco ENCODING= fv.json w.shp
vector convert --lco ENCODING=FOOBAR fv.json w.shp

vector convert --lco ENCODING=FOOBAR gonly.json w.shp
vector convert --lco ENCODING=1252 fv.json w.shp
vector convert --lco ENCODING=8859-1 fv.json w.shp
vector convert --lco ENCODING=1250 fc.json w.shp
vector convert fc.json w.shp
--config SHAPE_ENCODING=CP1251 vector convert fv.json w.shp
--config SHAPE_ENCODING=CP1251 vector convert --lco ENCODING=UTF-8 fv.json w.shp
--config SHAPE_ENCODING= vector convert fv.json w.shp
--config SHAPE_ENCODING=1250 vector convert fc.json w.shp

# append picks up the target layer encoding
vector convert --append ap.json t_cpg1251.shp
vector convert --append ap.json t_l87.shp
--config SHAPE_ENCODING=CP866 vector convert --append ap.json t_cpg1251.shp

# roundtrips
vector convert t_cpg1251.shp out.json
vector convert t_l87.shp out.json
